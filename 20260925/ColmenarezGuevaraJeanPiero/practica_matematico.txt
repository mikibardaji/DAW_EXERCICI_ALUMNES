import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        System.out.println("Start small. Ship something.");

        Scanner teclado = new Scanner(System.in);
        double monton_dinero, divisa, cambio_divisa;

        // Mostrar "Introduce monton dinero"
        System.out.println("Introduce monton de dinero:");
        // Esperar monton_dinero
        monton_dinero = teclado.nextDouble();

        // Mostrar "Precio divisa a cambiar"
        System.out.println("Precio divisa a cambiar:");
        // Esperar divisa
        divisa = teclado.nextDouble();

        // Calcular cambio_divisa = monton_dinero * divisa
        cambio_divisa = monton_dinero * divisa;

        // Mostrar "Tienes ", cambio_divisa
        System.out.println("Tienes " + cambio_divisa);

        teclado.close();
    }
}
